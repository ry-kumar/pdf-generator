const AWS = require('aws-sdk'); const PDFDocument = require('pdfkit'); const fs = require('fs'); // only needed if loading font from file locally const s3 = new AWS.S3(); exports.handler = async (event) => { try { const invoice = JSON.parse(event.body); // Create PDF in memory const doc = new PDFDocument({ margin: 50 }); const buffers = []; doc.on('data', buffers.push.bind(buffers)); doc.on('end', () => {}); // Register Times New Roman font doc.registerFont('TimesNewRoman', './Times-New-Roman.ttf'); doc.font('TimesNewRoman'); // Header doc.fontSize(20).text('Invoice', { align: 'center' }); doc.moveDown(); doc.fontSize(12) .text(`Invoice Number: ${invoice.number}`) .text(`Invoice Date: ${invoice.date}`) .text(`Customer: ${invoice.customerName}`) .moveDown(); // Table of items doc.fontSize(12).text('Items:', { underline: true }); doc.moveDown(0.5); invoice.items.forEach(item => { doc.text(`${item.description} - ${item.quantity} x $${item.price.toFixed(2)}`); }); // Subtotal, Tax, Discount, Total doc.moveDown(); doc.text(`Subtotal: $${invoice.subtotal.toFixed(2)}`); doc.text(`Tax: ${invoice.taxPercent.toFixed(2)}%`); doc.text(`Discount: ${invoice.discountPercent.toFixed(2)}%`); doc.moveDown(); doc.fontSize(14).text(`Total: $${invoice.total.toFixed(2)}`, { bold: true }); // Footer doc.moveDown(); doc.fontSize(12).text('Thank you for your business!', { align: 'center' }); doc.end(); // Convert buffers to single Buffer const pdfBuffer = await new Promise((resolve) => { const result = Buffer.concat(buffers); resolve(result); }); // Upload PDF to S3 const key = `invoices/invoice-${invoice.number}.pdf`; await s3.putObject({ Bucket: 'pdf-generator-12', Key: key, Body: pdfBuffer, ContentType: 'application/pdf' }).promise(); // Generate pre-signed URL const url = s3.getSignedUrl('getObject', { Bucket: 'pdf-generator-12', Key: key, Expires: 3600 }); return { statusCode: 200, body: JSON.stringify({ downloadUrl: url }) }; } catch (error) { console.error(error); return { statusCode: 500, body: 'Error generating invoice PDF' }; } };
